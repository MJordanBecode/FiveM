# 1. Crée la migration avec le compilateur 3.1 (Plus de problème de collation ou de Navigation !)
dotnet tool run dotnet-ef migrations add NomDeTaNouvelleMigration --no-build --project Data --startup-project Data

# 2. Supprime la DB
dotnet tool run dotnet-ef database drop --force --no-build --project Data --startup-project Data

# 3. Recrée la DB à jour
dotnet tool run dotnet-ef database update --no-build --project Data --startup-project Data
